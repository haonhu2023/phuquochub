# =============================================================================
# Authority Validation Rules
# =============================================================================
#
# Framework : Documentation Freeze
# Rule ID   : RULE-0003
# Category  : Authority
# Version   : 1.0.0
# Status    : Approved
#
# =============================================================================

metadata:

  id: RULE-0003

  name: Authority Validation

  category: Authority

  version: 1.0.0

  status: Approved

  owner: Architecture Governance

  description: >
    Validates authoritative engineering documentation and ensures
    exactly one approved source of truth exists for every mandatory domain.

# =============================================================================
# EXECUTION
# =============================================================================

execution:

  priority: 3

  blocking: true

  enabled: true

  stop_pipeline_on_failure: true

# =============================================================================
# PRECONDITIONS
# =============================================================================

preconditions:

  - repository_validation_completed

  - scope_validation_completed

# =============================================================================
# REQUIRED DOMAINS
# =============================================================================

mandatory_domains:

  - Architecture

  - ADR

  - Database

  - Workflow

  - API

  - Security

  - RBAC

  - Entity

# =============================================================================
# VALIDATION RULES
# =============================================================================

rules:

  - id: AUTH-001

    name: Authority Exists

    severity: Critical

    blocking: true

    condition:

      authority.exists: true

    on_fail:

      finding: No authority defined

      decision: FREEZE_REJECTED

  - id: AUTH-002

    name: Single Authority

    severity: Critical

    blocking: true

    condition:

      authority.count == 1

    on_fail:

      finding: Multiple authoritative documents detected

      decision: FREEZE_REJECTED

  - id: AUTH-003

    name: Authority Approved

    severity: Critical

    blocking: true

    condition:

      authority.status == Approved

    on_fail:

      finding: Authority not approved

      decision: FREEZE_REJECTED

  - id: AUTH-004

    name: Authority Current

    severity: High

    blocking: true

    condition:

      authority.superseded == false

    on_fail:

      finding: Superseded authority

      decision: FREEZE_REJECTED

  - id: AUTH-005

    name: Authority Version Defined

    severity: High

    blocking: true

    condition:

      authority.version_defined: true

    on_fail:

      finding: Authority version missing

      decision: FREEZE_REJECTED

  - id: AUTH-006

    name: Authority Owner Defined

    severity: High

    blocking: true

    condition:

      authority.owner_defined: true

    on_fail:

      finding: Authority owner undefined

      decision: FREEZE_REJECTED

  - id: AUTH-007

    name: Authority Scope Matches

    severity: High

    blocking: true

    condition:

      authority.scope_matches_repository: true

    on_fail:

      finding: Authority scope mismatch

      decision: FREEZE_REJECTED

  - id: AUTH-008

    name: Authority References Valid

    severity: Medium

    blocking: false

    condition:

      authority.references_valid: true

    on_fail:

      finding: Invalid authority references

  - id: AUTH-009

    name: Domain Coverage Complete

    severity: Critical

    blocking: true

    condition:

      authority.all_domains_covered: true

    on_fail:

      finding: Missing authoritative domain

      decision: FREEZE_REJECTED

  - id: AUTH-010

    name: Authority Hierarchy Valid

    severity: Critical

    blocking: true

    condition:

      authority.hierarchy_valid: true

    on_fail:

      finding: Invalid authority hierarchy

      decision: FREEZE_REJECTED

# =============================================================================
# HIERARCHY
# =============================================================================

authority_precedence:

  - Accepted ADR

  - Architecture Specification

  - Decision Register

  - Domain Specification

  - Repository Standard

  - Generated Documentation

  - README

# =============================================================================
# OUTPUT
# =============================================================================

outputs:

  produce:

    - authority_registry

    - authority_inventory

    - authority_graph

    - authority_status

    - authority_conflicts

# =============================================================================
# NEXT RULE
# =============================================================================

next:

  on_success: RULE-0004

  on_failure: STOP

# =============================================================================
# SUCCESS CRITERIA
# =============================================================================

success:

  authority_exists: true

  single_authority_per_domain: true

  authority_current: true

  authority_approved: true

  authority_hierarchy_valid: true

  continue_pipeline: true
