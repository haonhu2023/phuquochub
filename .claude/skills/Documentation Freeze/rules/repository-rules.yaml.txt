# =============================================================================
# Repository Validation Rules
# =============================================================================
#
# Framework : Documentation Freeze
# Rule ID   : RULE-0001
# Category  : Repository
# Version   : 1.0.0
# Status    : Approved
#
# =============================================================================

metadata:

  id: RULE-0001

  name: Repository Validation

  category: Repository

  version: 1.0.0

  status: Approved

  owner: Architecture Governance

  description: >
    Validates that the repository is accessible, complete,
    and suitable for Documentation Freeze.

# =============================================================================
# EXECUTION
# =============================================================================

execution:

  priority: 1

  blocking: true

  enabled: true

  stop_pipeline_on_failure: true

# =============================================================================
# PRECONDITIONS
# =============================================================================

preconditions:

  - repository_exists

  - repository_accessible

  - repository_readable

# =============================================================================
# VALIDATION RULES
# =============================================================================

rules:

  - id: REPO-001

    name: Repository Exists

    severity: Critical

    blocking: true

    condition:

      repository.exists: true

    on_fail:

      finding: Repository not found

      decision: FREEZE_BLOCKED

  - id: REPO-002

    name: Repository Readable

    severity: Critical

    blocking: true

    condition:

      repository.readable: true

    on_fail:

      finding: Repository cannot be read

      decision: FREEZE_BLOCKED

  - id: REPO-003

    name: Repository Root Exists

    severity: Critical

    blocking: true

    condition:

      repository.root_exists: true

    on_fail:

      finding: Repository root missing

      decision: FREEZE_BLOCKED

  - id: REPO-004

    name: Documentation Root Exists

    severity: Critical

    blocking: true

    condition:

      documentation.root_exists: true

    on_fail:

      finding: Documentation root missing

      decision: FREEZE_BLOCKED

  - id: REPO-005

    name: Documentation Accessible

    severity: High

    blocking: true

    condition:

      documentation.accessible: true

    on_fail:

      finding: Documentation inaccessible

      decision: FREEZE_BLOCKED

  - id: REPO-006

    name: Repository Version Defined

    severity: High

    blocking: true

    condition:

      repository.version_exists: true

    on_fail:

      finding: Repository version undefined

      decision: FREEZE_BLOCKED

  - id: REPO-007

    name: Repository Baseline Defined

    severity: High

    blocking: true

    condition:

      repository.baseline_exists: true

    on_fail:

      finding: Repository baseline undefined

      decision: FREEZE_BLOCKED

  - id: REPO-008

    name: Repository Integrity

    severity: Critical

    blocking: true

    condition:

      repository.integrity: valid

    on_fail:

      finding: Repository integrity failure

      decision: FREEZE_BLOCKED

  - id: REPO-009

    name: Repository Structure

    severity: Medium

    blocking: false

    condition:

      repository.structure_valid: true

    on_fail:

      finding: Repository structure inconsistent

  - id: REPO-010

    name: Required Documentation Folder

    severity: High

    blocking: true

    condition:

      repository.required_documentation_present: true

    on_fail:

      finding: Required documentation folders missing

      decision: FREEZE_BLOCKED

# =============================================================================
# OUTPUT
# =============================================================================

outputs:

  produce:

    - repository_inventory

    - repository_metadata

    - repository_version

    - repository_baseline

    - repository_status

# =============================================================================
# NEXT RULE
# =============================================================================

next:

  on_success: RULE-0002

  on_failure: STOP

# =============================================================================
# SUCCESS CRITERIA
# =============================================================================

success:

  repository_exists: true

  repository_accessible: true

  documentation_accessible: true

  repository_integrity_valid: true

  baseline_identified: true

  version_identified: true

  continue_pipeline: true
